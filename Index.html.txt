        window.renderHarian = function() {
            const tb = document.getElementById('harian-list'); if(!tb) return;
            const oF = document.getElementById('harFilterOutlet')?.value || ""; 
            const dF = document.getElementById('harFilterDept')?.value || "";
            const divF = document.getElementById('harFilterDivisi')?.value || window.currentDivisiFilter || "";
            const bF = parseInt(document.getElementById('harFilterBulan')?.value) || new Date().getMonth() + 1; 
            const tF = parseInt(document.getElementById('harFilterTahun')?.value) || new Date().getFullYear();

            // STRICTLY FILTER DAILIES ONLY (tanggal > 0) FOR HARIAN DASHBOARD
            let dD = window.data.filter(d => 
                (oF === "" || window.isOutletMatch(d.outlet, oF)) && 
                (dF === "" || d.dept === dF) && 
                (divF === "" || window.getDivisi(d.dept) === divF) &&
                d.bulan === bF && 
                d.tahun === tF &&
                d.tanggal > 0
            ).sort((a,b) => a.tanggal - b.tanggal);

            let tT = 0, tTP = 0, tTPem = 0;
            let tO = 0, tP = 0, tPem = 0; 
            let aD = []; let lSM = {}; let oLT = -1; let aH = {};
            
            let monthStockData = window.data.filter(d => 
                (oF === "" || window.isOutletMatch(d.outlet, oF)) && 
                (dF === "" || d.dept === dF) && 
                (divF === "" || window.getDivisi(d.dept) === divF) &&
                d.bulan === bF && 
                d.tahun === tF &&
                d.stock !== undefined && d.stock !== null && d.stock !== 0
            );

            monthStockData.forEach(d => {
                let kS_global = `${d.outlet}_${d.dept}`;
                if (!lSM[kS_global] || d.tanggal >= lSM[kS_global].tanggal) { 
                    lSM[kS_global] = { stock: d.stock, tanggal: d.tanggal, dept: d.dept, outlet: d.outlet }; 
                    if (d.tanggal > oLT) oLT = d.tanggal;
                }
            });

            dD.forEach(d => {
                tT += (d.target || 0); tTP += (d.tProfit || 0); tTPem += (d.tPembelian || 0);
                tO += (d.omset || 0); tP += (d.profit || 0); tPem += (d.pembelian || 0);
                
                if (!aD.includes(d.tanggal)) aD.push(d.tanggal);

                if (!aH[d.dept]) aH[d.dept] = { dept: d.dept, target: 0, omset: 0, tProfit: 0, profit: 0, tPem: 0, pem: 0, outlets: {}, links: [] };
                aH[d.dept].target += (d.target || 0); aH[d.dept].tProfit += (d.tProfit || 0); aH[d.dept].tPem += (d.tPembelian || 0);
                aH[d.dept].omset += (d.omset || 0); aH[d.dept].profit += (d.profit || 0); aH[d.dept].pem += (d.pembelian || 0);

                if (d.url && d.url.trim() !== "" && !aH[d.dept].links.find(l => l.url === d.url.trim())) aH[d.dept].links.push({ url: d.url.trim(), label: d.outlet });
            });

            // Populate stock into aH from lSM so department cards in Harian have accurate stock
            Object.values(lSM).forEach(sItem => {
                if (!aH[sItem.dept]) {
                    aH[sItem.dept] = { dept: sItem.dept, target: 0, omset: 0, tProfit: 0, profit: 0, tPem: 0, pem: 0, outlets: {}, links: [] };
                }
                let kS_dept = sItem.outlet || 'default';
                aH[sItem.dept].outlets[kS_dept] = { stock: sItem.stock, lastDate: sItem.tanggal };
            });

            // Historical Comparisons
            let pM = bF === 1 ? 12 : bF - 1; let pY = bF === 1 ? tF - 1 : tF; let pTY = tF - 1;
            
            let dL = window.data.filter(d => (oF === "" || window.isOutletMatch(d.outlet, oF)) && (dF === "" || d.dept === dF) && (divF === "" || window.getDivisi(d.dept) === divF) && d.bulan === pM && d.tahun === pY);
            let dTL = window.data.filter(d => (oF === "" || window.isOutletMatch(d.outlet, oF)) && (dF === "" || d.dept === dF) && (divF === "" || window.getDivisi(d.dept) === divF) && d.bulan === bF && d.tahun === pTY);

            let dbL = 0, dbPL = 0, dbTL = 0, dbPTL = 0;
            // Fetch from Master (tanggal 0) first, else fallback to sum of dailies
            let dLMasters = dL.filter(d => d.tanggal === 0);
            if(dLMasters.length > 0) { 
                dbL = dLMasters.reduce((sum, d) => sum + (d.omset||0), 0); 
                dbPL = dLMasters.reduce((sum, d) => sum + (d.profit||0), 0); 
            } else { 
                dbL = dL.reduce((sum, d) => sum + (d.omset||0), 0); 
                dbPL = dL.reduce((sum, d) => sum + (d.profit||0), 0); 
            }
            
            let dTLMasters = dTL.filter(d => d.tanggal === 0);
            if(dTLMasters.length > 0) { 
                dbTL = dTLMasters.reduce((sum, d) => sum + (d.omset||0), 0); 
                dbPTL = dTLMasters.reduce((sum, d) => sum + (d.profit||0), 0); 
            } else { 
                dbTL = dTL.reduce((sum, d) => sum + (d.omset||0), 0);
